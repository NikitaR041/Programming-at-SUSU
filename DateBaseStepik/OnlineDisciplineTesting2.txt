#1
В таблицу attempt включить новую попытку для студента Баранова Павла по дисциплине «Основы баз данных». Установить текущую дату в качестве даты выполнения попытки.
Для того, чтобы вставить текущую дату используйте функцию NOW().

Ответ:
INSERT INTO attempt (student_id, subject_id, date_attempt, result)
SELECT S.student_id, SU.subject_id, NOW(), NULL                
FROM student AS S
JOIN subject AS SU ON SU.name_subject = 'Основы баз данных'
WHERE S.name_student = 'Баранов Павел';

#2
Случайным образом выбрать три вопроса (запрос) по дисциплине, тестирование по которой собирается проходить студент, занесенный в таблицу attempt последним, и добавить их в таблицу testing. id последней попытки получить как максимальное значение id из таблицы attempt.

Ответ:
/*шаг 4 - отсортированная таблица, по которой будем делать выборку*/
SELECT Q.question_id
FROM question Q
WHERE Q.subject_id = (
    SELECT subject_id
    FROM attempt
    WHERE attempt_id = (SELECT MAX(ATT.attempt_id) FROM attempt ATT)
    )
ORDER BY RAND()
LIMIT 3;

INSERT INTO testing (attempt_id, question_id, answer_id)
SELECT 
    (SELECT MAX(attempt_id) FROM attempt) AS attempt_id,
    question_id, 
    NULL -- Можно без этого, тогда не нужно третьего атрибута в первой строке этого запроса
FROM (
    SELECT question_id
    FROM question
    WHERE subject_id = (
        SELECT subject_id
        FROM attempt
        WHERE attempt_id = (SELECT MAX(attempt_id) FROM attempt)
    )
    ORDER BY RAND()
    LIMIT 3
) AS random_questions;

select * from testing;
