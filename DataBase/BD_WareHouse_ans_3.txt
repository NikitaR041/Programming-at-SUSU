#1
Напишите функцию dbo.MonthSale (@year smallint, @month tinyint), возвращающую суммарную стоимость товара, проданного в году @year и месяце @month.

CREATE FUNCTION dbo.MonthSale(@year smallint, @month tinyint)
RETURNS money
AS
BEGIN
RETURN (
	SELECT ISNULL(SUM(SostNakl.Price * SostNakl.Amount),0)
	FROM SostNakl
	JOIN Nakl ON Nakl.Nakl_ID = SostNakl.Nakl_ID
	WHERE Nakl.InOut = '-' AND YEAR(Nakl.Dat) = @year AND MONTH(Nakl.Dat) = @month
	)
END

#2
Напишите функцию dbo.SumNakl9(@Nakl_ID int), возвращающую стоимость товаров по накладной с идентификатором @Nakl_ID. Из-за ошибок операторов накладная может не содержать товаров, тогда её стоимость - ноль.

CREATE FUNCTION dbo.SumNakl9(@Nakl_ID int)
RETURNS money
AS
BEGIN
RETURN (
	SELECT ISNULL(SUM(S.Price * S.Amount), 0)
        	FROM SostNakl S
        	WHERE S.Nakl_ID = @Nakl_ID
	)
END

Номер следующей по порядку исходящей накладной должен по умолчанию вычисляться как максимальный номер исходящей накладной плюс единица. Напишите функцию dbo.NextNumbNakl(), которая вычисляет этот номер.

CREATE FUNCTION dbo.NextNumbNakl()
RETURNS  int
AS
BEGIN
RETURN (
	SELECT MAX(Nakl.Numb)+1
	FROM Nakl
	WHERE Nakl.InOut = '-'
	)
END

CREATE FUNCTION dbo.NextNumbNakl()
RETURNS int
AS
BEGIN
    DECLARE @nextNumb int

    SELECT @nextNumb = ISNULL(MAX(Numb), 0) + 1
    FROM Nakl
    WHERE InOut = '-'

    RETURN @nextNumb
END

Напишите функцию dbo.Predok(@Tov_ID int), которая возвращает идентификатор корня дерева, в котором находится узел дерева с идентификатором @Tov_ID. Функцию можно написать, используя рекурсию, либо цикл, выполняющий подъём по ветвям дерева.

CREATE FUNCTION dbo.Predok(@Tov_ID int)
RETURNS int
AS
BEGIN
	DECLARE @MyParent int
	
	SELECT @MyParent = Tovar.Parent_ID
	FROM Tovar
	WHERE Tovar.Tovar_ID = @Tov_ID
	
	IF @MyParent IS NULL
		RETURN @Tov_ID

	RETURN dbo.Predok(@MyParent)
END

Напишите функцию 
TovarOnDate(@Tovar_ID int, @d datetime),
возвращающую количество товара @Tovar_ID на дату @d.
Примечание: контрольный запрос возвращает количество всех товаров на 01.01.2009.

CREATE FUNCTION dbo.TovarOnDate(@Tovar_ID int, @d datetime)
RETURNS INT
BEGIN

DECLARE @MyGlobalCount INT
DECLARE @TovarIn INT
DECLARE @TovarOut INT

SELECT @TovarIn = ISNULL(SUM(SN.Amount), 0)
FROM SostNakl SN
JOIN Nakl N ON N.Nakl_ID = SN.Nakl_ID
WHERE N.Dat > @d AND N.InOut LIKE '+' AND SN.Tovar_ID = @Tovar_ID

SELECT @TovarOut = ISNULL(SUM(SN.Amount), 0)
FROM SostNakl SN
JOIN Nakl N ON N.Nakl_ID = SN.Nakl_ID
WHERE N.Dat > @d AND N.InOut LIKE '-' AND SN.Tovar_ID = @Tovar_ID

SET @MyGlobalCount = @TovarOut - @TovarIn + (SELECT Amount 
					FROM Tovar
					WHERE Tovar_ID = @Tovar_ID)

RETURN @MyGlobalCount
END

Напишите функцию CurrentCostt(@Tovar_ID int),
которая возвращает текущую цену товара.
Примечание: поскольку нельзя использовать внутри функции недетерминированную функцию getdate(), используйте view CurrentDate, имеющееся в БД и возвращающее текущую дату. Например: select @curDate=CurDate from CurrentDate

create function dbo.CurrentCostt(@Tovar_ID int)
returns decimal(18, 4) as

begin

declare @CurDate datetime;
declare @CurCost decimal(18, 4);
select @CurDate = CurDate from CurrentDate
select @CurCost = Price
from PriceList
where Tovar_ID = @Tovar_ID
and @CurDate >= DateStart
and (@CurDate < DateEnd or DateEnd is null)
return @CurCost

end

ИЛИ

CREATE FUNCTION dbo.CurrentCostt(@Tovar_ID int)
RETURNS MONEY
BEGIN

DECLARE @CureDate DATE
SELECT @CureDate = CurDate from CurrentDate

RETURN (
	SELECT TOP 1 Price
	FROM PriceList PL
	WHERE PL.Tovar_ID = @Tovar_ID AND DateStart <= @CureDate AND (DateEnd IS NULL OR DateEnd >= @CureDate)
	ORDER BY DateStart DESC
	)
END
