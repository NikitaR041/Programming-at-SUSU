#1
Получить список пар названий организаций, расположенных по одному и тому же адресу. Для сравнения используйте идентификаторы организаций, а не имена. В первом столбце должна быть организация с меньшим идентификатором. Упорядочить сначала по названию первой организации из пары, а затем по второй. 

SELECT o1.OrgName AS Org1, o2.OrgName AS Org2
FROM Org o1
JOIN Org o2 ON o1.Address = o2.Address AND o1.Org_ID < o2.Org_ID
ORDER BY o1.OrgName, o2.OrgName

#2
Получить таблицу (дата закупки, номер накладной, количество, цена) для всех закупок товара 'Апельсин' в 2015 году. Упорядочить по дате закупки.

SELECT Dat, Numb, SostNakl.Amount, SostNakl.Price
FROM SostNakl 
JOIN Nakl ON Nakl.Nakl_ID = SostNakl.Nakl_ID
JOIN Tovar ON Tovar.Tovar_ID = SostNakl.Tovar_ID
WHERE Tovar.TovarName = 'Апельсин' AND Nakl.InOut = '+' AND YEAR(Nakl.Dat) = 2015
ORDER BY Nakl.Dat ASC

#3
Подсчитать число отпускных накладных (InOut='-') (OrgName, <число накладных>) для каждой организации в 2015 г. Упорядочить по названию организации.
Примечание: в ответе должны присутствовать и те организации, которые ничего не покупали со склада.

SELECT Org.OrgName, COUNT(Nakl.Nakl_ID)
FROM Org
LEFT JOIN Nakl ON Org.Org_ID = Nakl.Org_ID AND YEAR(Nakl.Dat) = 2015 AND Nakl.InOut = '-'
GROUP BY Org.OrgName
ORDER BY Org.OrgName ASC;

#5
Получить количество и стоимость проданного и купленного каждого из товаров в 2011 году для каждой организации (Организация, название товара, к-во проданного, стоимость проданного, к-во купленного, стоимость купленного). Упорядочить по названию организации, затем по названию товара. Если ничего не продавалось или не покупалось, то следует получить 0.

WITH DecartMultTable AS (
	SELECT O.OrgName, T.TovarName
	FROM Org O
	CROSS JOIN Tovar T
	WHERE T.IsTovar = 1
), MyTableSale AS (
	SELECT O.OrgName, T.TovarName, SUM(SN.Amount) AS CountA, SUM(SN.Amount * SN.Price) AS SaleSumTovars
	FROM Nakl N
	JOIN Org O ON O.Org_ID = N.Org_ID
	JOIN SostNakl SN ON SN.Nakl_ID = N.Nakl_ID
	JOIN Tovar T ON T.Tovar_ID = SN.Tovar_ID
	WHERE N.InOut LIKE '-' AND YEAR(N.Dat) = 2011 
	GROUP BY O.OrgName, T.TovarName
), MyTableBuy AS (
	SELECT O.OrgName, T.TovarName, SUM(SN.Amount) AS CountB, SUM(SN.Amount * SN.Price) AS BuySumTovars
	FROM Nakl N
	JOIN Org O ON O.Org_ID = N.Org_ID
	JOIN SostNakl SN ON SN.Nakl_ID = N.Nakl_ID
	JOIN Tovar T ON T.Tovar_ID = SN.Tovar_ID
	WHERE N.InOut LIKE '+' AND YEAR(N.Dat) = 2011
	GROUP BY O.OrgName, T.TovarName 
)

SELECT 
	DMT.OrgName AS Организация,
    	DMT.TovarName AS Название_товара,
    	COALESCE(MTS.CountA, 0) AS Количество_проданного,
    	COALESCE(MTS.SaleSumTovars, 0) AS Стоимость_проданного,
    	COALESCE(MTB.CountB, 0) AS Количество_купленного,
    	COALESCE(MTB.BuySumTovars, 0) AS Стоимость_купленного
FROM DecartMultTable DMT
LEFT OUTER JOIN MyTableSale MTS ON DMT.OrgName = MTS.OrgName AND DMT.TovarName = MTS.TovarName
LEFT OUTER JOIN MyTableBuy MTB ON DMT.OrgName = MTB.OrgName AND DMT.TovarName = MTB.TovarName
ORDER BY DMT.OrgName, DMT.TovarName

#6
Получить список (Название организации, Номер накладной, Дата накладной) для последних поступивших от организаций накладных (накладные на покупку). Упорядочить по названию организации, номеру накладной. Если накладных с максимальной датой для организации несколько, вывести их все.

SELECT OrgName, N.Numb, N.Dat
FROM Nakl N
JOIN Org O ON O.Org_ID = N.Org_ID
WHERE N.InOut LIKE '+' AND N.Dat = (SELECT MAX(N2.Dat) 
				FROM Nakl N2
				WHERE InOut LIKE '+' AND N2.Org_ID = N.Org_ID)
GROUP BY O.OrgName, N.Numb, N.Dat

#7
Найти противоречия в таблице PriceList, заключающиеся в пересечении сроков действия цены в виде (Название товара, Дата начала 1, Дата конца 1, Дата начала 2, Дата конца 2). Например, следующая запись обнаруживает противоречия: ('Помидоры', 01.01.2009, 15.01.2009, 10.01.2009, 25.01.2009). Результат упорядочить по названию товара и датам начала и конца первого и второго периодов.

SELECT T.TovarName, 
    	PL1.DateStart AS DateStart1, 
    	PL1.DateEnd AS DateEnd1, 
    	PL2.DateStart AS DateStart2, 
    	PL2.DateEnd AS DateEnd2
FROM PriceList PL1
JOIN PriceList PL2 ON PL1.PriceList_ID < PL2.PriceList_ID
JOIN Tovar T ON T.Tovar_ID = PL1.Tovar_ID
WHERE PL1.Tovar_ID = PL2.Tovar_ID AND NOT (PL1.DateEnd < PL2.DateStart OR PL2.DateEnd < PL1.DateStart)
ORDER BY T.TovarName, PL1.DateStart, PL1.DateEnd, PL2.DateStart, PL2.DateEnd

#9
Найти отпускную цену товара 'Апельсин' на 15.01.2015.
SELECT PL.Price
FROM PriceList PL
JOIN Tovar T ON T.Tovar_ID = PL.Tovar_ID 
WHERE T.TovarName LIKE 'Апельсин' AND '20150115' BETWEEN PL.DateStart AND PL.DateEnd
