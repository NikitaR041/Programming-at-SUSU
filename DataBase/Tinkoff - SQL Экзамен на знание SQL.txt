FROM plays P
JOIN castings C ON C.play_id = P.play_id
WHERE was_successful = 1
GROUP BY P.play_id, P.play_name
HAVING COUNT(*) = (SELECT MAX(cnt) 
                    FROM (SELECT COUNT(*) AS cnt
                        FROM castings 
                        WHERE was_successful = 1
                        GROUP BY play_id
                        )
                )

ИЛИ

WITH Successful_CTE
AS (
    SELECT play_id, COUNT(*) AS cnt
    FROM castings
    WHERE was_successful = 1
    GROUP BY play_id
)
WITH Max_successful_CTE
AS (
    SELECT MAX(cnt) AS m
    FROM Successful_CTE
)

SELECT P.play_name
FROM plays P
JOIN Successful_CTE S ON P.play_id = S.play_id
JOIN Max_successful_CTE MS ON S.cnt = MS.m;

1
SELECT emp_name, department, total_sales
FROM emp_sales_result
WHERE total_sales = max(total_sales);

2 +
SELECT emp_name, department, total_sales
FROM emp_sales_result AS t1
WHERE total_sales = (
    SELECT MAX(t2.total_sales)
    FROM emp_sales_result AS t2
    WHERE t1.department = t2.department
);

3
SELECT emp_name, department, total_sales
FROM emp_sales_result AS t1
WHERE department IN (
    SELECT MAX(t2.total_sales), department
    FROM emp_sales_result AS t2
    WHERE t1.department = t2.department
);

Какое из следующих утверждений о операторах UNION и UNION ALL является верным?

1)UNION возвращает только уникальные строки из объединяемых наборов данных, в то время как UNION ALL возвращает все строки, включая дубликаты. +
2)UNION работает быстрее, чем UNION ALL, так как не выполняет проверку на дубликаты.
3)Оба оператора требуют, чтобы объединяемые запросы имели одинаковое количество столбцов и соответствующие типы данных. +
4)UNION может использоваться только для объединения таблиц с одинаковым именем.
