#1
Найдите средний возраст сотрудников предприятия (в том числе и уволенных), выраженный в целых днях.

SELECT AVG(DATEDIFF(day, Birth, GETDATE()))

#2
Получите послужной список сотрудника 'Иванов Иван Владимирович' (Подразделение, должность, оклад, дата приказа, вид приказа). Упорядочить по дате приказа.

SELECT Depart.DepartName, JobTitle.JobTitleName, JobTitle.Salary, Move.Dat, MoveKind.MoveKindName
FROM Employee
LEFT JOIN Move ON Move.Emp_ID=Employee.Emp_ID
LEFT JOIN MoveKind ON MoveKind.MoveKind_ID = Move.MoveKind_ID
LEFT JOIN JobTitle ON JobTitle.JobTitle_ID = Move.JobTitle_ID
LEFT JOIN Depart ON Depart.Depart_ID = Move.Depart_ID
WHERE Employee.FIO = 'Иванов Иван Владимирович'
ORDER BY Move.Dat ASC

#3 
База данных содержит функцию dbo.curJobTitle (@Emp_ID int), которая возвращает идентификатор текущей должности, которую занимает сотрудник с идентификатором @Emp_ID. Используя её, постройте запрос, возвращающий фонд зарплаты для каждого подразделения (DepartName, Сумма). Упорядочить по названию подразделения.

SELECT DepartName, (SELECT ISNULL(SUM(Salary),0)
	FROM JobTitle, Employee
	WHERE dbo.curJobTitle(Emp_ID) = JobTitle.JobTitle_ID
	AND Depart.Depart_ID = Employee.Depart_ID)
FROM Depart
ORDER BY DepartName

#4
Для каждого подразделения найти подразделение, в которое оно входит. Ответ содержит поля: (название подразделения, название родителя). Упорядочить по названию подразделения.

SELECT d1.DepartName, d2.DepartName 
FROM Depart d1
LEFT OUTER JOIN Depart d2 ON d1.Parent_ID = d2.Depart_ID 
ORDER BY d1.DepartName

#5
Для каждого сотрудника найти количество его перемещений. Под перемещением понимается количество записей в таблице move. Упорядочить по фамилии сотрудника.

SELECT FIO , COUNT(Move.Move_ID)
FROM Employee
JOIN Move ON Move.Emp_ID = Employee.Emp_ID
GROUP BY FIO

#6
База данных содержит функцию dbo.curJobTitle (@Emp_ID int), которая возвращает идентификатор текущей должности, которую занимает сотрудник с идентификатором @Emp_ID. Получить список (должность, количество человек в этой должности) на текущую дату. В ответе должны присутствовать все должности, даже для тех должностей, которые никто не занимает. Упорядочить по названию должности.

SELECT JobTitleName, (SELECT COUNT(Emp_ID)
		FROM Employee
		WHERE dbo.curJobTitle(Emp_ID) = JobTitle.JobTitle_ID)
FROM JobTitle
ORDER BY JobTitleName ASC

#7
Количество принятых и переведенных в каждое подразделение в 2018 году (DepartName, Число сотрудников). Упорядочить по названию подразделения.

По идеи правильно так, но оказывается, нужно ещё учитывать не переведенных, тогда смотреть на "Решение 2"
"Решение 1"
SELECT D.DepartName, ISNULL(COUNT(MK.MoveKind_ID),0)
FROM Move M
JOIN MoveKind MK ON MK.MoveKind_ID = M.Move_ID
JOIN Depart D ON D.Depart_ID = M.Depart_ID
WHERE YEAR(M.Dat) = 2018
GROUP BY D.DepartName 
ORDER BY D.DepartName ASC

"Решение 2"
SELECT D.DepartName, COUNT(M.Move_ID)
FROM Depart D
LEFT JOIN Move M ON D.Depart_ID = M.Depart_ID AND YEAR(M.Dat) = 2018
LEFT JOIN MoveKind MK ON MK.MoveKind_ID = M.MoveKind_ID AND MK.IsArrival = 1
GROUP BY D.DepartName
ORDER BY D.DepartName ASC

#8
Список уволенных в период с 2016 по 2018 ('Уволен'). Упорядочить по фамилии.
SELECT E.FIO
FROM Employee E
LEFT JOIN Move M ON M.Emp_ID = E.Emp_ID
LEFT JOIN MoveKind MK ON MK.MoveKind_ID = M.MoveKind_ID
WHERE YEAR(M.Dat) >= 2016 AND YEAR(M.Dat) <= 2018 AND MK.IsArrival = 0
ORDER BY E.FIO ASC
