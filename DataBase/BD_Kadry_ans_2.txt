SET Salary = Salary * 2
WHERE Salary < 5000

Удалить все приказы из таблицы Move, имеющие дату (Dat) позже 01.01.2009.

#4
Уволить (MoveKindName='Уволен') из организации всех сотрудников, имеющих дату рождения ранее 15.11.1960. Дата увольнения - 21.01.2016.

INSERT INTO Move (Emp_ID, MoveKind_ID, Dat)
SELECT Emp_ID, 
	(SELECT MoveKind_ID FROM MoveKind WHERE MoveKindName = 'Уволен'), 
	'20160121'
FROM Employee
WHERE Birth < '19601115'

#6
Назначить сотрудника 'Яковлев Яков Яковлевич' на должность 'Директор' с 23.06.2016. MoveKind_ID=5 для назначения на должность, оставив его в текущем подразделении.

INSERT INTO Move (Emp_ID, MoveKind_ID, Dat, JobTitle_ID, Depart_ID)
SELECT E.Emp_ID, 5, '20160623', J.JobTitle_ID, E.Depart_ID        
FROM Employee E
JOIN JobTitle J ON J.JobTitleName = 'Директор'
WHERE E.FIO = 'Яковлев Яков Яковлевич';
