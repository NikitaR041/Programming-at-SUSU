#4
Все будущие рейсы компании 'Delta' перенести на неделю вперед.

UPDATE Flight
SET Dat = Dat + 7
FROM Company
WHERE Company.CompanyName = 'Delta' 
	AND Flight.Company_ID =Company.Company_ID 
	AND Dat > GETDATE()

#5
Рейс Челябинск - Магадан компании 'Delta' от 28.02.2015 выполнялся самолётом 'АН-2', а не 'Боинг-747', как это ошибочно указано в базе данных. Исправьте ошибку.

UPDATE Flight
SET PlaneType_ID = (
	SELECT PlaneType_ID FROM PlaneType WHERE PlaneTypeName = 'АН-2'
)
WHERE PlaneType_ID = (
	SELECT PlaneType_ID FROM PlaneType WHERE PlaneTypeName = 'Боинг-747'
)
AND Company_ID = (SELECT Company_ID FROM Company WHERE CompanyName = 'Delta')
AND CityFrom_ID = (SELECT City_ID FROM City WHERE CityName = 'Челябинск')
AND CityTo_ID = (SELECT City_ID FROM City WHERE CityName = 'Магадан')
AND DAY(Flight.Dat) = 28 AND MONTH(Flight.Dat) = 02 AND YEAR(Flight.Dat) = 2015


#6
Удалить из базы данных все города, в которые и из которых не летают никакие самолёты.

DELETE FROM City
WHERE NOT EXISTS(SELECT Flight_ID
FROM Flight
WHERE City.City_ID = Flight.CityFrom_ID OR City.City_ID = Flight.CityTo_ID)

#7
В тип самолёта 'А-310' (А - кириллица) добавить такие же посадочные места, которые имеются в 'ТУ-134' и которых там до этого не было.

INSERT INTO Seat (SeatName, PlaneType_ID)
SELECT S.SeatName, A310.PlaneType_ID
FROM Seat S
JOIN PlaneType A310 ON A310.PlaneTypeName LIKE 'А-310'
JOIN PlaneType TU134 ON TU134.PlaneTypeName LIKE 'ТУ-134'
WHERE S.PlaneType_ID = TU134.PlaneType_ID AND S.SeatName NOT IN (
		SELECT SeatName
		FROM Seat 
		WHERE PlaneType_ID = A310.PlaneType_ID)
