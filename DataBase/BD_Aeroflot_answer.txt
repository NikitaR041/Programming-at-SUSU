#1
Получить список названий компаний, названия которых начинаются на букву 'N'. Упорядочить по названию компании.

SELECT CompanyName
FROM Company
WHERE CompanyName LIKE 'N%'
ORDER BY CompanyName ASC

#2
Получить список полётов (город отправления, город назначения, дата, тип самолёта), выполненных после 2012 года компанией 'S7'. Упорядочить по полю Dat.

SELECT cFrom.CityName AS CityFrom, cTo.CityName AS CityTo, Flight.Dat, PlaneType.PlaneTypeName
FROM Flight
JOIN City cFrom ON Flight.CityFrom_ID = cFrom.City_ID
JOIN City cTo ON Flight.CityTo_ID = cTo.City_ID
JOIN PlaneType ON Flight.PlaneType_ID = PlaneType.PlaneType_ID
JOIN Company ON Flight.Company_ID = Company.Company_ID
WHERE YEAR(Flight.Dat) > 2012 AND Company.CompanyName = 'S7'
ORDER BY Flight.Dat ASC

#3
Список пассажиров, летавших на самолётах компании 'S7'. Упорядочить по алфавиту. Считать, что поле FIO уникально.

SELECT DISTINCT(FIO)
FROM Passgr
(LEFT)JOIN Flight ON Flight.Flight_ID = Passgr.Flight_ID
(LEFT)JOIN Company ON Company.Company_ID = Flight.Company_ID
WHERE Company.CompanyName = 'S7'
ORDER BY Passgr.FIO ASC

#4
Подсчитайте количество полётов (название города, количество полётов), совершённых из каждого города. Упорядочить по названию города.
Примечание: если из города не было полётов, он всё равно должен присутствовать в ответе (число полётов=0).

SELECT City.CityName, COUNT(Flight.Flight_ID)
FROM City
LEFT OUTER JOIN Flight ON Flight.CityFrom_ID = City.City_ID
GROUP BY CityName
ORDER BY CityName ASC

#5
Список пассажиров (FIO), когда-либо летавших в 'Санкт-Петербург' на самолете 'А-310' (буква А -кириллица). Упорядочить по алфавиту.
Примечание: предполагается, что FIO пассажиров уникальны.

SELECT DISTINCT(FIO)
FROM Passgr
JOIN Flight ON Flight.Flight_ID = Passgr.Flight_ID
JOIN City ON City.City_ID = Flight.CityTo_ID
JOIN PlaneType ON PlaneType.PlaneType_ID = Flight.PlaneType_ID
WHERE City.CityName LIKE 'Санкт-Петербург' AND PlaneType.PlaneTypeName LIKE 'А-310'
ORDER BY Passgr.FIO ASC

#6
Получить список типов самолётов, которыми выполняла рейсы авиакомпания 'S7'. Упорядочить по названию типа самолёта.

SELECT PlaneTypeName
FROM PlaneType
JOIN Flight ON Flight.PlaneType_ID = PlaneType.PlaneType_ID
JOIN Company ON Company.Company_ID = Flight.Company_ID
WHERE Company.CompanyName = 'S7'
GROUP BY PlaneTypeName
ORDER BY PlaneTypeName ASC

#7
Список городов, в которые никогда не выполнялись полёты. Упорядочить по алфавиту.

SELECT CityName
FROM City
WHERE NOT EXISTS (SELECT *
		FROM Flight
		WHERE City.City_ID = Flight.CityTo_ID)
ORDER BY City.CityName ASC

#8
Получить таблицу (название типа самолёта, число мест). Упорядочить по типу самолёта. Учтите, что в самолёте вообще может не быть мест (0 мест).

SELECT PT.PlaneTypeName, COUNT(S.Seat_ID)

#9
Получить список пассажиров (FIO), летевших из 'Санкт-Петербург' в 'Магадан' 27 апреля 2013 г. Упорядочить по FIO.

SELECT P.FIO
FROM Passgr P
JOIN Flight F ON F.Flight_ID = P.Flight_ID
WHERE DAY(F.Dat) = 27 AND MONTH(F.Dat) = 4 AND YEAR(F.Dat) = 2013
	AND F.CityFrom_ID = (SELECT C1.City_ID 
			FROM City C1
			WHERE C1.CityName LIKE 'Санкт-Петербург' 
				AND F.CityTo_ID = (SELECT C2.City_ID 
					FROM City C2
					WHERE C2.CityName LIKE 'Магадан')
			)
ORDER BY P.FIO ASC

#10
Список авиакомпаний, выполнявших когда-либо полёты из 'Санкт-Петербург' в 'Магадан'. Упорядочить по алфавиту.

SELECT C.CompanyName 
FROM Company C
JOIN Flight F ON F.Company_ID = C.Company_ID
WHERE F.CityFrom_ID = (SELECT C1.City_ID 
			FROM City C1
			WHERE C1.CityName = 'Санкт-Петербург' 
				AND F.CityTo_ID = (SELECT C2.City_ID
					FROM City C2
					WHERE C2.CityName = 'Магадан')
		         )
GROUP BY C.CompanyName 
ORDER BY C.CompanyName ASC 

#11

#14
Найдите тип(ы) самолёта(ов), имеющего(их) наибольшее количество мест. Примите во внимание, что таких может быть несколько. Вывести все. Упорядочить по типу самолёта.
Примечание: рекомендуется воспользоваться CTE (common table expression), но не обязательно.

WITH name1(ID, n) AS
(
SELECT PlaneType_ID, count(Seat_ID)
FROM Seat
GROUP BY PlaneType_ID
)

SELECT PlaneTypeName
FROM PlaneType, name1
WHERE PlaneType.PlaneType_ID = name1.ID AND n = (SELECT max(n) FROM name1)
