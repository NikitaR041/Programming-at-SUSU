#1
Организация 'Рога и копыта' сменила название. Теперь она называется 'Копыта и рога'. Внесите это изменение в базу данных.

UPDATE Org
SET OrgName = 'Копыта и рога'
WHERE OrgName = 'Рога и копыта'

#2
Увеличить количество товара на складе 'Черешня' на 20 кг.

UPDATE Tovar
SET Amount = Amount + 20
WHERE (SELECT Tovar_ID FROM Tovar WHERE TovarName = 'Черешня' ) = Tovar_ID
