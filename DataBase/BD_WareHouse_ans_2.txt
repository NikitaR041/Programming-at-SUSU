#1
Организация 'Рога и копыта' сменила название. Теперь она называется 'Копыта и рога'. Внесите это изменение в базу данных.

UPDATE Org
SET OrgName = 'Копыта и рога'
WHERE OrgName = 'Рога и копыта'

#2
Увеличить количество товара на складе 'Черешня' на 20 кг.

UPDATE Tovar
SET Amount = Amount + 20
WHERE (SELECT Tovar_ID FROM Tovar WHERE TovarName = 'Черешня' ) = Tovar_ID

#3
Товар 'Мандарины' перенести в категорию 'Фрукты'.

UPDATE Tovar
SET Parent_ID = (
	SELECT Tovar_ID
	FROM Tovar
	WHERE TovarName = 'Фрукты')
WHERE TovarName = 'Мандарины'

#4
Добавить в таблицу PriceList запись о новой цене на 'Черешня', равной 600 руб. с 01.03.2016.

INSERT INTO PriceList(Tovar_ID, Price, DateStart)
VALUES ((SELECT Tovar_ID FROM Tovar WHERE TovarName = 'Черешня' AND IsTovar = 1), 600, '20160301')

#5
Удалить все исходящие накладные, в которых отсутствует список товаров.

DELETE FROM Nakl
WHERE InOut LIKE '-' AND Nakl_ID NOT IN(SELECT Nakl_ID FROM SostNakl)

#6
Удалить из таблицы Tovar все товары, которые ни разу ни покупались, ни продавались и на которые никогда не назначалась цена.

DELETE FROM Tovar
WHERE IsTovar = 1
	AND Tovar_ID NOT IN(SELECT DISTINCT Tovar_ID FROM SostNakl)
	AND Tovar_ID NOT IN(SELECT DISTINCT Tovar_ID FROM PriceList)
