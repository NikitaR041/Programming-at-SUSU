#1
Создайте таблицу dbo.ttt со следующими свойствами. Поля:
a - целое, 4 байта, может принимать неопределенные значения;
b - символьное переменной длины, 1 байт на символ, не более 20 символов, всегда определено, значение по умолчанию 'Привет';
n - целое, 2 байта, автоинкрементное, первичный ключ, начальное значение 2 шаг инкремента 2;
d - вычисляемое по формуле a+n хранимое поле.

CREATE TABLE dbo.ttt (
	a int NULL,
	b varchar(20) NOT NULL DEFAULT 'Привет',
	n smallint IDENTITY(2,2) PRIMARY KEY,
	d AS (a+n)	 PERSISTED  
)
